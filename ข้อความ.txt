            // --- CHART VARIABLES ---
            let overviewOsChart = null;

            const htmlEl = document.documentElement;
            
            function getChartOptions(isDark, type = 'doughnut') {
                const textColor = isDark ? 'white' : 'black';
                const gridColor = isDark ? '#333333' : '#E0E0E0';
                
                let options = {
                    maintainAspectRatio: false,
                    responsive: true,
                    plugins: { legend: { labels: { color: textColor } }, tooltip: { enabled: true } },
                    scales: {
                        y: { ticks: { color: textColor }, grid: { color: gridColor } },
                        x: { ticks: { color: textColor }, grid: { color: gridColor } }
                    }
                };

                if (type === 'doughnut' || type === 'pie') { options.scales = {}; options.plugins.tooltip.enabled = true; } 
                else if (type === 'bar' || type === 'line') { options.scales.y.beginAtZero = true; options.plugins.tooltip.enabled = true; }
                
                return options;
            }
            
            function createChart(ctx, type, data, labels, datasets = null) {
                const isDark = htmlEl.classList.contains('dark');
                const chartOptions = getChartOptions(isDark, type);
                
                let chartConfig = { type: type, data: {}, options: chartOptions };

                if (type === 'doughnut') {
                    chartConfig.data = { labels: labels, datasets: [{ label: 'Market Share', data: data, backgroundColor: ['#007BFF', '#28A745', '#FFC107', '#DC3545', '#6C757D', '#17A2B8'], }] };
                } else if (type === 'line') {
                    chartConfig.data = { labels: labels, datasets: datasets.map(d => ({ ...d, pointBackgroundColor: d.borderColor, pointBorderColor: '#fff', pointHoverBackgroundColor: '#fff', pointHoverBorderColor: d.borderColor, tension: 0.4 })) };
                } else if (type === 'bar') {
                    chartConfig.data = { labels: labels, datasets: [{ label: 'Popularity (%)', data: data, backgroundColor: 'rgba(0, 123, 255, 0.7)', borderColor: 'rgba(0, 123, 255, 1)', borderWidth: 1 }] };
                }
                
                return new Chart(ctx, chartConfig);
            }
            
            function updateChartTheme(chartInstance) {
                if (!chartInstance) return;
                const isDark = htmlEl.classList.contains('dark');
                const type = chartInstance.config.type;
                const newOptions = getChartOptions(isDark, type);
                
                chartInstance.options.plugins.legend.labels.color = newOptions.plugins.legend.labels.color;
                
                if (chartInstance.options.scales.x) { 
                    chartInstance.options.scales.x.ticks.color = newOptions.scales.x.ticks.color; 
                    chartInstance.options.scales.x.grid.color = newOptions.scales.x.grid.color; 
                }
                if (chartInstance.options.scales.y) { 
                    chartInstance.options.scales.y.ticks.color = newOptions.scales.y.ticks.color; 
                    chartInstance.options.scales.y.grid.color = newOptions.scales.y.grid.color; 
                }

                chartInstance.update();
            }

            });

            // --- SORTING ALGORITHM LOGIC ---
            const sortChartCtx = document.getElementById('sort-chart')?.getContext('2d');
            let dataArray = [];
            const arraySize = 30;
            let isSorting = false;

            if (sortChartCtx) {
                const pseudocodes = {
                    bubble: `BubbleSort(A, n):\n  for i from 1 to n-1\n    for j from 0 to n-i-1\n      if A[j] > A[j+1]\n        swap(A[j], A[j+1])`,
                    selection: `SelectionSort(A, n):\n  for i from 0 to n-1\n    minIndex = i\n    for j from i+1 to n-1\n      if A[j] < A[minIndex]\n        minIndex = j\n    swap(A[i], A[minIndex])`,
                    insertion: `InsertionSort(A, n):\n  for i from 1 to n-1\n    key = A[i]\n    j = i - 1\n    while j >= 0 and A[j] > key\n      A[j+1] = A[j]\n      j = j - 1\n    A[j+1] = key`
                };
                
                const pseudocodeEl = document.querySelector('#pseudocode pre');
                const algoSelect = document.getElementById('algo-select');
                const sortBtn = document.getElementById('sort-btn');
                const resetBtn = document.getElementById('reset-btn');

                function updatePseudocode() {
                    pseudocodeEl.textContent = pseudocodes[algoSelect.value];
                }
                algoSelect.addEventListener('change', updatePseudocode);

                function createRandomArray() {
                    dataArray = Array.from({length: arraySize}, () => Math.floor(Math.random() * 95) + 5);
                }

                function initSortChart() {
                    if (sortChart) sortChart.destroy();
                    sortChart = new Chart(sortChartCtx, {
                        type: 'bar',
                        data: {
                            labels: dataArray.map((_, i) => i + 1),
                            datasets: [{
                                data: dataArray,
                                backgroundColor: '#d1d5db',
                                borderColor: '#f9fafb',
                                borderWidth: 2
                            }]
                        },
                        options: {
                            responsive: true,
                            maintainAspectRatio: false,
                            plugins: { legend: { display: false }, tooltip: { enabled: false } },
                            scales: { y: { display: false, min: 0 }, x: { display: false } },
                            animation: { duration: 0 }
                        }
                    });
                }

                function updateChart(highlights = {}) {
                    if (!sortChart) return;
                    sortChart.data.datasets[0].data = [...dataArray];
                    sortChart.data.datasets[0].backgroundColor = dataArray.map((_, i) => {
                        if (highlights.sorted && i < highlights.sorted) return '#10b981';
                        if (highlights.final && highlights.final.includes(i)) return '#10b981';
                        if (highlights.current === i) return '#f59e0b'; // Yellow (current element being checked)
                        if (highlights.compare === i) return '#ef4444'; // Red (element being compared to)
                        if (highlights.key === i) return '#007BFF'; // Blue (key or minimum element)
                        return '#d1d5db'; // Default gray
                    });
                    sortChart.update('none');
                }
                
                function setUIState(sorting) {
                    isSorting = sorting;
                    sortBtn.disabled = sorting;
                    resetBtn.disabled = sorting;
                    algoSelect.disabled = sorting;
                    sortBtn.textContent = sorting ? 'Sorting...' : 'Sort';
                }

                function reset() {
                    setUIState(false);
                    createRandomArray();
                    initSortChart();
                    updatePseudocode();
                }

                resetBtn.addEventListener('click', reset);
                
                const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));
                
                async function sort() {
                    if (isSorting) return;
                    setUIState(true);

                    const algo = algoSelect.value;
                    const speed = 50;
                    const n = dataArray.length;

                    if (algo === 'bubble') {
                        for (let i = 0; i < n - 1; i++) {
                            for (let j = 0; j < n - i - 1; j++) {
                                if(!isSorting) { setUIState(false); return; }
                                updateChart({ current: j, compare: j + 1, final: Array.from({length: i}, (_, k) => n - 1 - k) });
                                await sleep(speed);
                                if (dataArray[j] > dataArray[j + 1]) {
                                    [dataArray[j], dataArray[j + 1]] = [dataArray[j + 1], dataArray[j]];
                                }
                            }
                        }
                    } else if (algo === 'selection') {
                        for (let i = 0; i < n; i++) {
                            let minIndex = i;
                            for (let j = i + 1; j < n; j++) {
                                if(!isSorting) { setUIState(false); return; }
                                updateChart({ current: j, key: minIndex, sorted: i });
                                await sleep(speed/2);
                                if (dataArray[j] < dataArray[minIndex]) {
                                    minIndex = j;
                                }
                            }
                            [dataArray[i], dataArray[minIndex]] = [dataArray[minIndex], dataArray[i]];
                            updateChart({ key: i, sorted: i + 1 });
                            await sleep(speed);
                        }
                    } else if (algo === 'insertion') {
                        for (let i = 1; i < n; i++) {
                            let key = dataArray[i];
                            let j = i - 1;
                            updateChart({ key: i, sorted: i });
                            await sleep(speed);
                            while (j >= 0 && dataArray[j] > key) {
                                if(!isSorting) { setUIState(false); return; }
                                dataArray[j + 1] = dataArray[j];
                                updateChart({ current: j, compare: j+1, key: i, sorted: i });
                                await sleep(speed);
                                j--;
                            }
                            dataArray[j + 1] = key;
                        }
                    }
                    
                    updateChart({ final: Array.from({length: n}, (_, k) => k) });
                    setUIState(false);
                }

                sortBtn.addEventListener('click', sort);
                reset();
            }
        });
